  RAT MAZE :

  #include<bits/stdc++.h>
  using namespace std;
  int n;
  int maze[10][10],sol[10][10];
  bool issafe(int x,int y)
  {
      if(x>=0 && x<n && y>=0 && y<n && maze[x][y]==1)
      {
          return true;
      }
      return false;
  }
  bool solvemazeutil(int x,int y)
  {
      if(x==n-1 && y==n-1 && maze[x][y]==1)
      {
          sol[x][y]=1;
          return true;
      }
      if(issafe(x,y)==true)
      {
          if(sol[x][y]==1)
          {
              return false;
          }
          sol[x][y]=1;
          if(solvemazeutil(x + 1,y)==true)
            return true;
          if(solvemazeutil(x,y + 1)==true)
            return true;
          sol[x][y]=0;
          return false;
      }
      return false;
  }
  void solvemaze()
  {
      for(int i=0;i<n;i++)
      {
          for(int j=0;j<n;j++)
          {
              sol[i][j]=0;
          }
      }
      if(solvemazeutil(0,0)==false)
      {
          cout<<"No solution";
      }
      else
      {
          for(int i=0;i<n;i++)
          {
              for(int j=0;j<n;j++)
              {
                  cout<<sol[i][j]<<" ";
              }
              cout<<endl;
          }
      }
  }
  int main()
  {
      cin>>n;
      for(int i=0;i<n;i++)
      {
          for(int j=0;j<n;j++)
          {
              cin>>maze[i][j];
          }
      }
      solvemaze();
  }
